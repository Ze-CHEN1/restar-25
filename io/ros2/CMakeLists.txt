cmake_minimum_required(VERSION 3.16)
project(restar_ros_publishers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(restar_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_cmake_pytest REQUIRED)

message(STATUS "Building Restar ROS communication tools: no camera/CAN/serial dependencies. Full sentry is NOT built.")
add_library(restar_ros_publishers STATIC publish2nav.cpp)
target_include_directories(restar_ros_publishers PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(restar_ros_publishers Eigen3::Eigen Threads::Threads)
ament_target_dependencies(restar_ros_publishers rclcpp std_msgs robot_interfaces)
add_library(restar_ros_io STATIC ros2.cpp subscribe2nav.cpp)
target_link_libraries(restar_ros_io restar_ros_publishers)
ament_target_dependencies(restar_ros_io rclcpp restar_msgs)
add_executable(restar_ros_mock mock_publisher.cpp)
target_link_libraries(restar_ros_mock restar_ros_publishers)
add_executable(restar_ros_lifecycle_test runtime_lifecycle_test.cpp)
target_link_libraries(restar_ros_lifecycle_test restar_ros_publishers)
add_executable(restar_ros_inputs_test inputs_test.cpp)
target_link_libraries(restar_ros_inputs_test restar_ros_io)

# MC02 protocol/transport live in io/mc02_serial; this ROS adapter owns only
# subscriptions, cache policy and periodic frame publication.
add_library(mc02_protocol STATIC ../mc02_serial/src/protocol.cpp)
target_include_directories(mc02_protocol PUBLIC ../mc02_serial/include)
add_library(mc02_serial_sender STATIC ../mc02_serial/src/serial_sender.cpp)
target_include_directories(mc02_serial_sender PUBLIC ../mc02_serial/include)
target_link_libraries(mc02_serial_sender Threads::Threads)
add_executable(mc02_serial_bridge_node mc02_bridge/bridge_node.cpp)
target_include_directories(mc02_serial_bridge_node PRIVATE ../mc02_serial/include)
target_link_libraries(mc02_serial_bridge_node mc02_protocol mc02_serial_sender)
ament_target_dependencies(mc02_serial_bridge_node rclcpp geometry_msgs robot_interfaces
  std_msgs diagnostic_msgs rcl_interfaces)
add_executable(mc02_mock_receiver mc02_bridge/mock_receiver.cpp)
target_include_directories(mc02_mock_receiver PRIVATE ../mc02_serial/include)
target_link_libraries(mc02_mock_receiver mc02_protocol Threads::Threads)
ament_target_dependencies(mc02_mock_receiver rclcpp std_msgs)
add_executable(mc02_frame_dump ../mc02_serial/src/frame_dump.cpp)
target_include_directories(mc02_frame_dump PRIVATE ../mc02_serial/include)
target_link_libraries(mc02_frame_dump mc02_protocol)

include(CTest)

  add_executable(mc02_protocol_test ../mc02_serial/test/protocol_test.cpp)
  target_include_directories(mc02_protocol_test PRIVATE ../mc02_serial/include)

  target_link_libraries(mc02_protocol_test mc02_protocol)
  add_test(NAME mc02_protocol_test COMMAND mc02_protocol_test)

  add_executable(mc02_crc_compat_test ../mc02_serial/test/crc_compat_test.cpp ../../tools/crc.cpp)
  target_include_directories(mc02_crc_compat_test PRIVATE ../mc02_serial/include ../../tools)
  target_link_libraries(mc02_crc_compat_test mc02_protocol)
  add_test(NAME mc02_crc_compat_test COMMAND mc02_crc_compat_test)

  add_executable(mc02_serial_sender_test ../mc02_serial/test/serial_sender_test.cpp)
  target_include_directories(mc02_serial_sender_test PRIVATE ../mc02_serial/include)
  target_link_libraries(mc02_serial_sender_test mc02_serial_sender mc02_protocol)
  add_test(NAME mc02_serial_sender_test COMMAND mc02_serial_sender_test)

  ament_add_pytest_test(mc02_bridge_runtime mc02_bridge/test_bridge_runtime.py TIMEOUT 120)
  ament_add_pytest_test(mc02_serial_pty mc02_bridge/test_serial_pty.py TIMEOUT 60)
  set(MC02_TEST_ENV "ROS_DOMAIN_ID=83;ROS_LOCALHOST_ONLY=1;PYTHONNOUSERSITE=1;ROS_LOG_DIR=/tmp/restar_mc02_test_logs;MC02_BRIDGE_BIN_DIR=$<TARGET_FILE_DIR:mc02_serial_bridge_node>")
  set_tests_properties(mc02_bridge_runtime mc02_serial_pty PROPERTIES
    ENVIRONMENT "${MC02_TEST_ENV}")
endif()
